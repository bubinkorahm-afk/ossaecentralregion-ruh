Google Sheets & Cloud Sync

Connect a free Google Sheet to receive student exam attempts live from their phones or computers.

How to set up a free Google Sheet (1 minute)
  1. Create a new spreadsheet in Google Sheets.
  2. Click Extensions → Apps Script.
  3. Paste the script below and click Deploy → New Deployment → Select type: Web App.
  4. Set Who has access to Anyone, then click Deploy and copy the URL here.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Timestamp", "Class", "Student Name", "Roll No", "Phone", "Sunday School", "School Code", "Diocese", "Score", "Total", "Percentage", "Status", "Duration"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.id, data.timestamp, data.className, data.studentName,
    data.rollNumber, data.studentPhone, data.schoolName || '', data.schoolCode, data.diocese,
    data.score, data.totalQuestions, data.percentage + '%', data.status, data.durationStr
  ]);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
