How to set up or update the Google Sheet script
  1. Open your spreadsheet in Google Sheets (or create one at sheets.new). Copy its ID from the address bar — the long part between /d/ and /edit.
  2. Click Extensions → Apps Script (or open your project at script.google.com). Delete all the old code and paste the script below. Then change two lines at the top:
    • READ_KEY: replace CHANGE-ME-to-a-secret-word with your own secret word (keep the quote marks).
    • SHEET_ID: paste your Sheet ID between the quote marks (needed for a project made at script.google.com).
    Click 💾 Save (Ctrl + S) and wait until it has saved.
  3. First time: click Deploy → New deployment, type Web app, Execute as: Me, Who has access: Anyone, then Deploy and allow access.
    Updating an existing script: click Deploy → Manage deployments → ✏️ Edit → Version: New version → Deploy. The URL stays the same.
  4. Paste the Web app URL above (it ends with /exec), enter your secret word as the read key, and click Save & Test. For students’ results to reach the Sheet, the same URL must also be in site-config.js as sheetUrl.
// OSSAE Exam Portal — Google Sheet connector (exam results + feedback)
var READ_KEY = 'CHANGE-ME-to-a-secret-word';  // your secret word — the admin page asks for it

// Only needed if this is a standalone project (created at script.google.com):
// paste the ID from your Google Sheet's address, the long part between /d/ and /edit
var SHEET_ID = '';

var ATTEMPT_HEADERS = ['ID', 'Timestamp', 'Date', 'Time', 'Class', 'Student Name', 'Roll No', 'Phone',
  'Sunday School', 'School Code', 'Diocese', 'Score', 'Total', 'Percentage', 'Status', 'Duration'];
var FEEDBACK_HEADERS = ['ID', 'Timestamp', 'Name', 'Phone', 'Class', 'Rating', 'Message', 'Page', 'Language'];

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    if (d.type === 'feedback') {
      sheet_('Feedback', FEEDBACK_HEADERS).appendRow([
        text_(d.id), text_(d.timestamp), text_(d.name), text_(d.phone), text_(d.className),
        d.rating || '', text_(String(d.message || '').slice(0, 1000)), text_(d.page), text_(d.lang)
      ]);
    } else {
      var sh = sheet_('Attempts', ATTEMPT_HEADERS);
      var row = [
        text_(d.id), text_(d.timestamp), text_(d.dateStr), text_(d.timeStr), text_(d.className),
        text_(d.studentName), text_(d.rollNumber), text_(d.studentPhone), text_(d.schoolName),
        text_(d.schoolCode), text_(d.diocese), Number(d.score) || 0, Number(d.totalQuestions) || 0,
        Number(d.percentage) || 0, text_(d.status), text_(d.durationStr)
      ];
      // One row per attempt: update it when the exam finishes or is aborted
      var found = sh.getRange(1, 1, sh.getLastRow(), 1)
        .createTextFinder(String(d.id)).matchEntireCell(true).findNext();
      if (found && found.getRow() > 1) sh.getRange(found.getRow(), 1, 1, row.length).setValues([row]);
      else sh.appendRow(row);
    }
    return json_({status: 'ok'});
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.key !== READ_KEY) return json_({error: 'unauthorized'});
  var isFeedback = p.type === 'feedback';
  var sh = book_().getSheetByName(isFeedback ? 'Feedback' : 'Attempts');
  var rows = sh ? sh.getDataRange().getDisplayValues().slice(1) : [];
  return json_(rows.map(function(r) {
    return isFeedback
      ? {id: r[0], timestamp: r[1], name: r[2], phone: r[3], className: r[4], rating: r[5],
         message: r[6], page: r[7], lang: r[8]}
      : {id: r[0], timestamp: r[1], dateStr: r[2], timeStr: r[3], className: r[4], studentName: r[5],
         rollNumber: r[6], studentPhone: r[7], schoolName: r[8], schoolCode: r[9], diocese: r[10],
         score: Number(r[11]) || 0, totalQuestions: Number(r[12]) || 0, percentage: Number(r[13]) || 0,
         status: r[14], durationStr: r[15]};
  }));
}

function book_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (ss) return ss;
  if (!SHEET_ID) throw new Error('No Google Sheet linked: set SHEET_ID at the top of the script.');
  return SpreadsheetApp.openById(SHEET_ID);
}

function sheet_(name, headers) {
  var ss = book_();
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);
  if (sh.getLastRow() === 0) sh.appendRow(headers);
  return sh;
}

// Store as plain text; stops "+966..." becoming a number and blocks formula injection
function text_(v) {
  v = (v === undefined || v === null) ? '' : String(v);
  return /^[=+\-@]/.test(v) ? "'" + v : v;
}

function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
